Send structured questions to Jev through `/v1/systemone` and verify the answers. JavaScript and Python use the official Jev SDKs with the Inference SDK as their transport; Rust and Go use native NEAR AI clients.

## Send and verify a decision

Set `NEARAI_API_KEY` and choose a System One model from the [catalog](/cloud/models).

Keep E2EE, OHTTP, request compression, and automatic retries disabled as shown below. The official Jev clients use `https://cloud-api.near.ai` as their base URL, without `/v1`.

<Tabs>
<Tab title="JavaScript">

Requires Node.js 24.22.1 or later.

```bash
npm install @nearai/inference-sdk @typesafe-ai/sdk
```

```javascript
import { InferenceClient } from "@nearai/inference-sdk/node";
import { TypeSafeClient, noul } from "@typesafe-ai/sdk";

const apiKey = process.env.NEARAI_API_KEY;
if (!apiKey) throw new Error("Set NEARAI_API_KEY first.");

const inference = new InferenceClient({
  apiKey,
  baseUrl: "https://cloud-api.near.ai/v1",
  e2ee: false,
  ohttp: false,
});
const jev = new TypeSafeClient({
  apiKey,
  baseURL: "https://cloud-api.near.ai",
  fetch: inference.fetch,
  compression: false,
  retry: { maxRetries: 0 },
});

const { data, response } = await jev.systemOne({
  model: "typesafe/jev-1.13",
  state: { message: "I was charged twice. Please help." },
  questions: { billing: noul("Is this about billing?") },
}).withResponse();

// Use the NEAR receipt ID, which is separate from Jev's requestId.
const id = response.headers.get("X-Generation-Id");
if (!id) throw new Error("Missing System One receipt ID.");
const verified = await inference.verifyResponse(id);
console.log(data.answers);
console.log("Verified signer:", verified.signatureKind);
```

<Tab title="Python">

Requires Python 3.12 or later.

```bash
pip install nearai-inference-sdk typesafe-sdk
```

```python
import asyncio
import os

from nearai_inference_sdk import InferenceClient
from typesafe_sdk import AsyncTypeSafeClient, Noul, RetryPolicy


async def main():
    api_key = os.environ["NEARAI_API_KEY"]
    async with InferenceClient(
        api_key,
        base_url="https://cloud-api.near.ai/v1",
        e2ee=False,
        ohttp=False,
    ) as inference:
        async with AsyncTypeSafeClient(
            api_key=api_key,
            base_url="https://cloud-api.near.ai",
            http_client=inference.httpx2_client,
            compression=False,
            retry=RetryPolicy(max_retries=0),
        ) as jev:
            result = await jev.system_one(
                model="typesafe/jev-1.13",
                state={"message": "I was charged twice. Please help."},
                questions={"billing": Noul(instructions="Is this about billing?")},
            )
            # Use the NEAR receipt ID, not Jev's request_id.
            receipt_id = result.raw_http_response.headers.get("X-Generation-Id")
            if not receipt_id:
                raise RuntimeError("Missing System One receipt ID.")
            verified = await inference.verify_response(receipt_id)
            print(result.answers)
            print("Verified signer:", verified.signature_kind)


asyncio.run(main())
```

cargo add nearai-inference-sdk

```rust
use nearai_inference_sdk::{InferenceClient, InferenceClientOptions};
use serde_json::json;

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let client = InferenceClient::with_options(InferenceClientOptions {
        api_key: Some(std::env::var("NEARAI_API_KEY")?),
        base_url: "https://cloud-api.near.ai/v1".into(),
        e2ee: false,
        ohttp: false,
        ..Default::default()
    })?;
    let result = client.systemone_create(&json!({
        "model": r#"typesafe/jev-1.13"#,
        "state": {"message": "I was charged twice. Please help."},
        "questions": {
            "billing": {"type": "noul", "instructions": "Is this about billing?"}
        }
    })).await?;

    // decision_id is captured from X-Generation-Id.
    let verified = client.verify_response(&result.decision_id).await?;
    println!("{}", serde_json::to_string_pretty(&result.data.answers)?);
    println!("Verified signer: {:?}", verified.signature_kind());
    Ok(())
}
```

```go
package main

import (
    "context"
    "encoding/json"
    "fmt"
    "log"
    "os"
    "time"

    nearai "github.com/nearai/inference-sdk/go"
)

func main() {
    if err := run(); err != nil {
        log.Fatal(err)
    }
}

func run() error {
    apiKey := os.Getenv("NEARAI_API_KEY")
    if apiKey == "" {
        return fmt.Errorf("set NEARAI_API_KEY first")
    }
    ctx, cancel := context.WithTimeout(context.Background(), 2*time.Minute)
    defer cancel()
    client, err := nearai.NewInferenceClient(nearai.InferenceOptions{
        ClientOptions: nearai.ClientOptions{
            APIKey: apiKey,
            BaseURL: "https://cloud-api.near.ai/v1",
        },
        E2EE: false,
        OHTTP: false,
    })
    if err != nil {
        return err
    }
    defer client.Close()

    result, err := client.CreateSystemOne(ctx, nearai.SystemOneRequest{
        Model: "typesafe/jev-1.13",
        State: map[string]string{"message": "I was charged twice. Please help."},
        Questions: map[string]nearai.SystemOneQuestion{
            "billing": {Type: "noul", Instructions: "Is this about billing?"},
        },
    })
    if err != nil {
        return err
    }
    // DecisionID is captured from X-Generation-Id.
    verified, err := client.VerifyResponse(ctx, result.DecisionID)
    if err != nil {
        return err
    }
    answers, err := json.MarshalIndent(result.Data.Answers, "", "  ")
    if err != nil {
        return err
    }
    fmt.Println(string(answers))
    fmt.Println("Verified signer:", verified.SignatureKind)
    return nil
}
```

## Receipt verification

Use the `X-Generation-Id` response header to verify the decision. This is separate from Jev's request ID. The native Rust and Go clients expose it as `decision_id` and `DecisionID`.

If the receipt is temporarily unavailable, retry verification with the same ID. Response records expire after 60 minutes by default, so verify before expiry or closing the client. See [Verification](/cloud/inference-sdk#verification) for the meaning of each signature kind.

## More examples

System One supports JSON requests with `noul` (probability), `choice`, and `score` questions. See the [SDK examples](https://github.com/nearai/inference-sdk/tree/main/examples) for more question types and native JavaScript/Python clients.

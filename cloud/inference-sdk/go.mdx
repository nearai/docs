## Install

Requires Go **1.26.10 or later**. Install the published SDK in your application:

```bash
go mod init example.com/my-app # If your project does not have a go.mod yet.
go get github.com/nearai/inference-sdk/go
export NEARAI_API_KEY="your-api-key"
```

## Examples

Choose the tab matching your model's API. Save the example as `main.go` and run it with `go run .`.

<Tab title="Chat Completions" id="chat-completions">

Call a supported NEAR TEE model with E2EE enabled, then verify the completion.

```go
package main

import (
	"context"
	"encoding/json"
	"fmt"
	"log"
	"os"
	"time"

	nearai "github.com/nearai/inference-sdk/go"
)

func main() {
	if err := run(); err != nil {
		log.Fatal(err)
	}
}

func run() error {
	apiKey := os.Getenv("NEARAI_API_KEY")
	if apiKey == "" {
		return fmt.Errorf("set NEARAI_API_KEY")
	}
	ctx, cancel := context.WithTimeout(context.Background(), 2*time.Minute)
	defer cancel()

	client, err := nearai.NewInferenceClient(nearai.InferenceOptions{
		ClientOptions: nearai.ClientOptions{APIKey: apiKey},
		E2EE:          true,
	})
	if err != nil {
		return err
	}
	defer client.Close()

	body, err := client.CreateChatCompletion(ctx, map[string]any{
		"model":    "z-ai/glm-5.3-flash",
		"messages": []map[string]string{{"role": "user", "content": "Hello!"}},
	})
	if err != nil {
		return err
	}
	var completion struct {
		ID string `json:"id"`
	}
	if err := json.Unmarshal(body, &completion); err != nil {
		return err
	}
	if completion.ID == "" {
		return fmt.Errorf("completion has no ID")
	}

	verified, err := client.VerifyResponse(ctx, completion.ID)
	if err != nil {
		return err
	}
	fmt.Println(verified.SignatureKind)
	fmt.Println(string(body))
	return nil
}
```

<Tab title="Responses" id="responses">

Use a Responses model with `store: false` and E2EE disabled. See [Stateless Responses](/cloud/guides/stateless-responses) for conversation history and tool calls.

```bash
go get github.com/openai/openai-go/v3
```

```go
package main

import (
    "context"
    "fmt"
    "log"
    "os"
    "time"

    nearai "github.com/nearai/inference-sdk/go"
    "github.com/openai/openai-go/v3"
    "github.com/openai/openai-go/v3/option"
    "github.com/openai/openai-go/v3/responses"
)

func main() {
    if err := run(); err != nil {
        log.Fatal(err)
    }
}

func run() error {
    apiKey := os.Getenv("NEARAI_API_KEY")
    if apiKey == "" {
        return fmt.Errorf("set NEARAI_API_KEY first")
    }
    ctx, cancel := context.WithTimeout(context.Background(), 2*time.Minute)
    defer cancel()

    client, err := nearai.NewInferenceClient(nearai.InferenceOptions{
        ClientOptions: nearai.ClientOptions{
            APIKey: apiKey,
            BaseURL: "https://cloud-api.near.ai/v1",
        },
        E2EE: false,
    })
    if err != nil {
        return err
    }
    defer client.Close()

    ai := openai.NewClient(
        option.WithAPIKey(apiKey),
        option.WithBaseURL(client.BaseURL()),
        option.WithHTTPClient(client.HTTPClient()),
        option.WithMaxRetries(0),
    )
    response, err := ai.Responses.New(ctx, responses.ResponseNewParams{
        Model: "openai/gpt-6-astra",
        Input: responses.ResponseNewParamsInputUnion{OfString: openai.String("Hello!")},
        Store: openai.Bool(false),
    })
    if err != nil {
        return err
    }
    if response.ID == "" {
        return fmt.Errorf("response has no ID")
    }

    // Verify before displaying the answer.
    verified, err := client.VerifyResponse(ctx, response.ID)
    if err != nil {
        return err
    }
    fmt.Println(response.OutputText())
    fmt.Println("Verified signer:", verified.SignatureKind)
    return nil
}
```

Call an Anthropic model through the Messages API with E2EE disabled.

```go
package main

import (
    "context"
    "encoding/json"
    "fmt"
    "log"
    "os"
    "time"

    nearai "github.com/nearai/inference-sdk/go"
)

func main() {
    if err := run(); err != nil {
        log.Fatal(err)
    }
}

func run() error {
    apiKey := os.Getenv("NEARAI_API_KEY")
    if apiKey == "" {
        return fmt.Errorf("set NEARAI_API_KEY first")
    }
    ctx, cancel := context.WithTimeout(context.Background(), 2*time.Minute)
    defer cancel()
    client, err := nearai.NewInferenceClient(nearai.InferenceOptions{
        ClientOptions: nearai.ClientOptions{
            APIKey: apiKey,
            BaseURL: "https://cloud-api.near.ai/v1",
        },
        E2EE: false,
    })
    if err != nil {
        return err
    }
    defer client.Close()

    raw, err := client.CreateMessage(ctx, map[string]any{
        "model": "anthropic/claude-opus-4-6",
        "max_tokens": 128,
        "messages": []map[string]string{{"role": "user", "content": "Hello!"}},
    })
    if err != nil {
        return err
    }
    var response struct {
        ID string
        Content []struct { Type, Text string }
    }
    if err := json.Unmarshal(raw, &response); err != nil {
        return err
    }
    if response.ID == "" {
        return fmt.Errorf("message has no ID")
    }

    // Verify before displaying the answer.
    verified, err := client.VerifyResponse(ctx, response.ID)
    if err != nil {
        return err
    }
    for _, block := range response.Content {
        if block.Type == "text" {
            fmt.Println(block.Text)
        }
    }
    fmt.Println("Verified signer:", verified.SignatureKind)
    return nil
}
```

Ask Jev a structured question with E2EE and OHTTP disabled.

```go
package main

import (
    "context"
    "encoding/json"
    "fmt"
    "log"
    "os"
    "time"

    nearai "github.com/nearai/inference-sdk/go"
)

func main() {
    if err := run(); err != nil {
        log.Fatal(err)
    }
}

func run() error {
    apiKey := os.Getenv("NEARAI_API_KEY")
    if apiKey == "" {
        return fmt.Errorf("set NEARAI_API_KEY first")
    }
    ctx, cancel := context.WithTimeout(context.Background(), 2*time.Minute)
    defer cancel()
    client, err := nearai.NewInferenceClient(nearai.InferenceOptions{
        ClientOptions: nearai.ClientOptions{
            APIKey: apiKey,
            BaseURL: "https://cloud-api.near.ai/v1",
        },
        E2EE: false,
        OHTTP: false,
    })
    if err != nil {
        return err
    }
    defer client.Close()

    result, err := client.CreateSystemOne(ctx, nearai.SystemOneRequest{
        Model: "typesafe/jev-1.13",
        State: map[string]string{"message": "I was charged twice. Please help."},
        Questions: map[string]nearai.SystemOneQuestion{
            "billing": {Type: "noul", Instructions: "Is this about billing?"},
        },
    })
    if err != nil {
        return err
    }
    // DecisionID is captured from X-Generation-Id.
    verified, err := client.VerifyResponse(ctx, result.DecisionID)
    if err != nil {
        return err
    }
    answers, err := json.MarshalIndent(result.Data.Answers, "", "  ")
    if err != nil {
        return err
    }
    fmt.Println(string(answers))
    fmt.Println("Verified signer:", verified.SignatureKind)
    return nil
}
```

## E2EE

Gateway E2EE is disabled by default. Set `E2EE: true` to encrypt supported Chat fields to an attested model key and decrypt the response automatically.

E2EE requires a NEAR vLLM model with attestation enabled. Other providers use Gateway-only verification; leave E2EE disabled for those models.

## Verification

Before sending Chat, the client verifies the Gateway and every returned model report for supported models. Attested Gateway TLS pinning is enabled by default; normal HTTPS certificate validation remains enabled. Failed required checks stop the request.

Response-signature verification is explicit: call `client.VerifyResponse(ctx, completionID)` after fully consuming the response. The client retains exact request and response bytes automatically, including encrypted wire bytes when E2EE is enabled.

See [Verification](/cloud/inference-sdk#verification) for receipt IDs and signature kinds.

Response records expire 60 minutes after the full response is read, by default. Verify each response before its record expires or the client closes.

To check a deployment without sending a prompt, call `client.Verify(ctx, model)`. For custom policies and cache settings, see the [Go verification guide](https://github.com/nearai/inference-sdk/blob/main/go/docs/verification-guide.md).

## Use the OpenAI Go SDK

`InferenceClient` implements `http.RoundTripper`. To use an existing official OpenAI Go client, pass the verified HTTP client and disable automatic completion retries:

```bash
go get github.com/openai/openai-go/v3
```

```go
// Imports:
// "github.com/openai/openai-go/v3"
// "github.com/openai/openai-go/v3/option"

ai := openai.NewClient(
    option.WithAPIKey(apiKey),
    option.WithBaseURL(client.BaseURL()),
    option.WithHTTPClient(client.HTTPClient()),
    option.WithMaxRetries(0),
)
```

Use `client` from the completion example and the same API key. The transport supports Chat Completions and Responses, including streaming. Disable E2EE for Responses.

For streaming, consume the stream, check `stream.Err()`, and close it before calling `client.VerifyResponse(ctx, completionID)`. Use a completion ID from the stream. See the [OpenAI streaming example](https://github.com/nearai/inference-sdk/blob/main/examples/example-go/gateway/openai/main.go).

## OHTTP

Set `OHTTP: true` in `InferenceOptions` to encapsulate the Chat HTTP exchange to the attested Gateway. OHTTP is disabled by default and requires Ed25519, the default signing algorithm. Configure E2EE independently. Missing or invalid Gateway OHTTP evidence blocks Chat.

<Note>
  OHTTP supports Chat Completions, Responses, and Messages through the Gateway; System One does not support OHTTP. Attestation and signature requests remain regular HTTPS. The endpoint must expose `/ohttp` at the same origin. Outer authorization headers and the client's network address are not hidden by OHTTP.
</Note>
